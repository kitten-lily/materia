# All three containers join this pod and share one network namespace, so they
# reach each other over localhost (traefik/gerbil configs point at localhost:3001).
# 127.0.0.1). Ports are published here, at the pod level — NOT on the individual
# containers. This is required: Gerbil creates WireGuard tunnel interfaces in
# the shared namespace, and Traefik must reach the tunnel endpoint IPs
# (100.89.137.0/20 CGNAT range) to proxy traffic to tunneled resources.
[Unit]
Description=Pangolin edge pod
Wants=network-online.target
After=network-online.target

[Pod]
PodName=pangolin
# Keep the infra container alive when every member container is stopped.
# Quadlet's DEFAULT is ExitPolicy=stop, which drains the pod the moment the
# last member exits — and because gerbil.service and traefik.service both
# Requires=app.service, systemd stops ALL THREE whenever app is restarted.
# The pod then stops itself, and app's restart fails with "Bound to unit
# pangolin-pod.service, but unit isn't active" → total edge outage from an
# ordinary app.container change (BUG-009, hit 2026-09-21). With `continue`
# the infra container (and therefore the shared netns + published ports)
# survives member restarts.
ExitPolicy=continue
PublishPort=80:80/tcp
PublishPort=443:443/tcp
PublishPort=443:443/udp
PublishPort=51820:51820/udp
PublishPort=21820:21820/udp
# Buildbarn raw TCP resources (bst-cache-index / bst-cache-storage on bow)
# — see components/buildbarn/blueprint-resources.yaml and
# specs/plans/issue-28-bst-cache-krytis.md. Raw TCP/UDP resources need a
# manual Gerbil port mapping per Pangolin's own "Proxied Resources" docs;
# there's no dynamic-resource mechanism for these like HTTP resources get.
PublishPort=7981:7981/tcp
PublishPort=7982:7982/tcp

[Service]
Restart=always

[Install]
WantedBy=multi-user.target default.target
